#vault-squirreldoc 
test case for
- 新的分区与已存在的分区时间范围不重叠的场景
#### ddl
```sql
DROP TABLE if exists nafp_bcccps_bou_day;
DROP TABLE if exists nafp_bcccps_bou_day_1;
DROP TABLE if exists nafp_bcccps_bou_day_2;
DROP TABLE if exists nafp_bcccps_bou_day_3;
DROP TABLE if exists nafp_bcccps_bou_day_4;

CREATE TABLE public.nafp_bcccps_bou_day (
    d_record_id integer NOT NULL,
    d_datetime numeric
)
PARTITION BY RANGE (d_datetime);

CREATE TABLE public.nafp_bcccps_bou_day_1 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
ALTER TABLE ONLY public.nafp_bcccps_bou_day ATTACH PARTITION public.nafp_bcccps_bou_day_1 FOR VALUES FROM ('2022') TO ('2023');

CREATE TABLE public.nafp_bcccps_bou_day_2 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
ALTER TABLE ONLY public.nafp_bcccps_bou_day ATTACH PARTITION public.nafp_bcccps_bou_day_2 FOR VALUES FROM ('2023') TO ('2025');

CREATE TABLE public.nafp_bcccps_bou_day_3 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
CREATE TABLE public.nafp_bcccps_bou_day_4 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
```

#### test 
- session 1
```sql
begin;
insert into nafp_bcccps_bou_day values(1, 2022);
end;
```

- session2
```sql
begin;
alter table nafp_bcccps_bou_day detach partition nafp_bcccps_bou_day_2;

ALTER TABLE ONLY public.nafp_bcccps_bou_day ATTACH PARTITION public.nafp_bcccps_bou_day_3 FOR VALUES FROM ('2023') TO ('2024');
ALTER TABLE ONLY public.nafp_bcccps_bou_day ATTACH PARTITION public.nafp_bcccps_bou_day_4 FOR VALUES FROM ('2024') TO ('2025');
end;
```

- session3

```sql
insert into nafp_bcccps_bou_day values(100, 2022);
insert into nafp_bcccps_bou_day values(101, 2023);
insert into nafp_bcccps_bou_day values(102, 2024);

select*from nafp_bcccps_bou_day_1;
select*from nafp_bcccps_bou_day_3;
select*from nafp_bcccps_bou_day_4;
```
