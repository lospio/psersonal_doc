- 新的分区与已存在的分区时间范围重叠的场景
#### ddl
```sql
DROP TABLE if exists nafp_bcccps_bou_day;
DROP TABLE if exists nafp_bcccps_bou_day_1;
DROP TABLE if exists nafp_bcccps_bou_day_2;
DROP TABLE if exists nafp_bcccps_bou_day_3;
DROP TABLE if exists nafp_bcccps_bou_day_4;

CREATE TABLE public.nafp_bcccps_bou_day (
    d_record_id integer NOT NULL,
    d_datetime numeric
)
PARTITION BY RANGE (d_datetime);

CREATE TABLE public.nafp_bcccps_bou_day_1 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
ALTER TABLE ONLY public.nafp_bcccps_bou_day ATTACH PARTITION public.nafp_bcccps_bou_day_1 FOR VALUES FROM ('2022') TO ('2025');

CREATE TABLE public.nafp_bcccps_bou_day_2 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);

CREATE TABLE public.nafp_bcccps_bou_day_3 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);
CREATE TABLE public.nafp_bcccps_bou_day_4 (
    d_record_id integer NOT NULL,
    d_datetime numeric
);

insert into nafp_bcccps_bou_day (select generate_series(1,100), 2022);
```

```sql
-- Set boundaries

tatest :='2021-06-01'::DATE;

NOT VALID;

-- You should not touch pg_catalog directly
UPDATE pg_constraint

END;
```
